BUG FIX 1: Admin approve/reject buttons not working
==================================================

FILE: src/screens/admin/AdminDashboard.tsx

ROOT CAUSE (confirmed from code review):
The reviewGym and reviewEdit functions use Alert.alert with confirmation dialogs.
In some Expo environments, Alert.alert does not reliably fire the onPress callback,
causing the buttons to appear to do nothing.

The code structure is actually correct — the issue is ONLY the Alert.alert wrapper.

FIX:
Remove Alert.alert from reviewGym and reviewEdit entirely.
Execute the state update directly on button press without confirmation dialog.
Instead show inline visual feedback after the action.

Step 1: Replace the reviewGym function:

CURRENT (wrong):

    Alert.alert(
      action === 'approved' ? '헬스장 승인' : '헬스장 거절',
      ...,
      [
        { text: '취소', style: 'cancel' },
        {
          text: action === 'approved' ? '승인' : '거절',
          onPress: () => {
            setPendingGyms(prev => {
              const updated = prev.map(g => g.id === gymId ? { ...g, status: action } : g);
              const updatedGym = updated.find(g => g.id === gymId);
              if (updatedGym) setSelectedGym(updatedGym);
              return updated;
            });
          },
        },
      ],
    );
  }, []);

NEW (correct — no Alert):
  const reviewGym = useCallback((gymId: string, action: ReviewStatus) => {
    setPendingGyms(prev => {
      const updated = prev.map(g => g.id === gymId ? { ...g, status: action } : g);

    // src/data/managedGymStore.ts
    interface ManagedGymEntry {
      gymId: string
      machineIds: string[]
    }

    const gymMachineData: ManagedGymEntry[] = []

    export function getGymMachineIds(gymId: string): string[] {
      return gymMachineData.find(e => e.gymId === gymId)?.machineIds ?? []
    }

    export function updateGymMachines(gymId: string, machineIds: string[]): void {
      const existing = gymMachineData.find(e => e.gymId === gymId)
      if (existing) {
        existing.machineIds = [...machineIds]
      } else {
        gymMachineData.push({ gymId, machineIds: [...machineIds] })
      }
    }
