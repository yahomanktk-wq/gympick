/**
 * 헬스장의 머신 목록을 Supabase에 저장한다.
 * 기존 gym_machines 레코드를 삭제 후 새로 insert (upsert 방식).
 */
export async function saveGymMachineIds(gymId: string, machineIds: string[]): Promise<{ error: string | null }> {
  try {
    const { error: deleteError } = await supabase
      .from('gym_machines')
      .delete()
      .eq('gym_id', gymId);
    if (deleteError) return { error: deleteError.message };

    if (machineIds.length === 0) return { error: null };

    const rows = machineIds.map(machineId => ({
      gym_id:     gymId,
      machine_id: machineId,
      unit_count: 1,
      condition:  'good',
    }));

    const { error: insertError } = await supabase
      .from('gym_machines')
      .insert(rows);
    if (insertError) return { error: insertError.message };

    return { error: null };
  } catch (e: any) {
    return { error: e.message ?? '알 수 없는 오류' };
  }
}

■ 종합 — Supabase 전환 시 작업 포인트 정리 (수정 지시 아님, 전환 준비 참고용)
================================================================================

1. [필수] gyms 테이블에 직접등록 식별 컬럼 추가 필요
   현재 is_admin_registered 등 플래그 없음. manager_id=null로는 식별 불가.
   추가 없이는 Supabase에서 직접등록 헬스장만 조회하는 쿼리를 작성할 수 없음.

2. [필수] handleRegisterGym + handleUpdateGym에 Supabase gym_machines 저장 추가
   현재 saveGymMachineIds 미호출 → gym_machines 누락.
   매니저 패턴(saveGymMachineIds 직접 호출) 그대로 재활용 가능.

3. [필수] handleUpdateGym에 supabase.gyms.update 추가
   현재 인메모리만 갱신. saveEdits 패턴(updatePayload 구조) 재활용 가능.

4. [필수] 목록 조회(registeredGyms state 초기화 + 갱신)를 Supabase 기반으로 전환
   관련 지점: L1368 useState 초기화, L1610/L1657 setRegisteredGyms 갱신.
