  Get gym machines from the existing gym data:

        .map(gm => gm.machineId),
      [gymId]

  Get user favorites:

    const favoriteMachineIds = getUserFavorites('u1')

  Build filter options dynamically:

      const unique = [...new Set(gymMachines.map(m => m.targetMuscle).filter(Boolean))].sort()
      return ['전체', ...unique]
    }, [gymMachines])

    const brandOptions = useMemo(() => {
      const unique = [...new Set(gymMachines.map(m => m.brand).filter(Boolean))].sort()
      return ['전체', ...unique]
    }, [gymMachines])

  Filter machines:
    const filteredMachines = useMemo(() => {
      return gymMachines.filter(m => {
        const muscleOk = machineMuscleFilter === '전체' ||
          m.targetMuscle?.toLowerCase() === machineMuscleFilter.toLowerCase()
        const brandOk = machineBrandFilter === '전체' ||
          m.brand?.toLowerCase() === machineBrandFilter.toLowerCase()
        const favOk = !showFavoritesOnly || favoriteMachineIds.includes(m.id)
        return muscleOk && brandOk && favOk
      })
    }, [gymMachines, machineMuscleFilter, machineBrandFilter, showFavoritesOnly])

  Render machine tab UI:
    <ScrollView>
      {/* 즐겨찾기 필터 토글 */}
      <View style={favFilterRow}>
        <TouchableOpacity
          style={[favChip, !showFavoritesOnly && favChipActive]}
          onPress={() => setShowFavoritesOnly(false)}
        >
          <Text>전체</Text>
        </TouchableOpacity>
        <TouchableOpacity
          style={[favChip, showFavoritesOnly && favChipActive]}
          onPress={() => setShowFavoritesOnly(true)}
        >
          <Ionicons name="heart" size={13} color={showFavoritesOnly ? '#000' : C.textSub} />
          <Text>즐겨찾기</Text>
        </TouchableOpacity>
      </View>

      {/* 타겟부위 필터 */}
      <ScrollView horizontal showsHorizontalScrollIndicator={false}>
        {muscleOptions.map(opt => (
          <TouchableOpacity
            key={opt}
            style={[filterChip, machineMuscleFilter === opt && filterChipActive]}
            onPress={() => setMachineMuscleFilter(opt)}
          >
            <Text>{opt}</Text>
          </TouchableOpacity>
        ))}
      </ScrollView>

      {/* 브랜드 필터 */}
      <ScrollView horizontal showsHorizontalScrollIndicator={false}>
        {brandOptions.map(opt => (
          <TouchableOpacity
            key={opt}
            style={[filterChip, machineBrandFilter === opt && filterChipActive]}
            onPress={() => setMachineBrandFilter(opt)}
          >
            <Text>{opt}</Text>
          </TouchableOpacity>
        ))}
      </ScrollView>

      {/* 머신 카드 그리드 */}
      {filteredMachines.length === 0 ? (
        <View style={emptyState}>
          <Text>조건에 맞는 머신이 없어요</Text>
        </View>
      ) : (
        <View style={machineGrid}>
          {filteredMachines.map(machine => (
            <TouchableOpacity
              key={machine.id}
              onPress={() => {
                setSelectedMachine(machine)
                setMachineDetailVisible(true)
              }}
            >
              <MachineCard machine={machine} variant="grid" width={CARD_W} />
            </TouchableOpacity>
          ))}
        </View>
      )}
    </ScrollView>

  Add machine detail modal inside GymDetailModal:
