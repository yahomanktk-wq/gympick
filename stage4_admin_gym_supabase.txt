================================================================================
[4단계] 어드민 직접등록 헬스장 — Supabase 영구저장 전환
================================================================================

■ 작업 개요
어드민 '헬스장 직접 등록'을 인메모리(adminGymStore)에서 Supabase로 전환한다.
- 등록/수정/머신연결/목록조회 전부 Supabase 기반 (앱 꺼도 유지)
- 보유 머신이 gym_machines 테이블에 저장됨 (현재 버그 수정)
- 헬스장 정보/머신 수정이 Supabase에 반영됨 (현재 버그 수정)
- 직접등록 헬스장 목록에 검색 기능 추가
- 등록일(created_at) 표시
- manager_id는 null 유지 (직접등록은 '직접 관리' 배지 안 뜸 — 의도된 동작)

■ 직접등록 식별 방법
gyms 테이블에서 manager_id IS NULL 인 헬스장 = 어드민 직접등록.
(확인 완료: 다른 모든 경로는 manager_id를 채우거나 gyms를 생성하지 않음)
  · 어드민 직접등록 → manager_id null
  · 매니저 등록요청 승인 → manager_id = 매니저 id (null 아님)
  · 매니저 신청 승인 → gyms 생성 안 함
  · 유저 등록 제안 → gyms 생성 안 함 (status만 변경)
※ 향후 '유저 제안 → gyms 자동생성' 기능을 추가하면 이 식별이 깨질 수 있으므로,
  그때 source 컬럼 도입을 재검토할 것. (지금은 불필요)

import { supabase } from '../supabase';
import { saveGymMachineIds } from './gymMachines';
import { invalidateGymsCache } from './gyms';

// 어드민이 직접 등록한 헬스장 (manager_id IS NULL)
export interface AdminGym {
  id:         string;
  name:       string;
  location:   string;
  phone:      string;
  hours:      string;
  machineIds: string[];
  createdAt:  string;
}

// 직접등록 헬스장 목록 조회 (manager_id IS NULL + 보유 머신)
export async function fetchAdminGyms(): Promise<AdminGym[]> {
  const { data, error } = await supabase
    .from('gyms')
    .select(`
      id, name, location, phone, hours, created_at,
      manager_id,
      gym_machines ( machine_id )
    `)
    .is('manager_id', null)
    .order('created_at', { ascending: false });

  if (error || !data) {
    if (__DEV__ && error) console.warn('[db/adminGyms] fetchAdminGyms error:', error.message);
    return [];
  }

  return data.map((row: any) => ({
    id:         row.id,
    name:       row.name,
    location:   row.location,
    phone:      row.phone ?? '',
    hours:      row.hours ?? '',
    machineIds: (row.gym_machines ?? []).map((gm: any) => gm.machine_id),
    createdAt:  row.created_at ?? '',
  }));
}

// 직접등록 (gyms insert + 머신 저장)
export async function insertAdminGym(
  name: string, location: string, phone: string, hours: string, machineIds: string[],
): Promise<{ error: string | null }> {
  try {
    const gymId = `g_${Date.now()}`;
    const { geocodeAddress } = await import('../geocoding');
    const coords = await geocodeAddress(location.trim());

    const { error: insertError } = await supabase.from('gyms').insert({
      id:          gymId,
      name:        name.trim(),
      location:    location.trim(),
      phone:       phone.trim(),
      hours:       hours.trim(),
      description: '',
      latitude:    coords?.latitude  ?? null,
      longitude:   coords?.longitude ?? null,
      manager_id:  null,
    });
    if (insertError) return { error: insertError.message };

    // 보유 머신 저장 (gym_machines)
    const { error: machineError } = await saveGymMachineIds(gymId, machineIds);
    if (machineError) return { error: machineError };

    invalidateGymsCache();
    return { error: null };
  } catch (e: any) {
    return { error: e.message ?? '알 수 없는 오류' };
  }
}

// 직접등록 헬스장 수정 (gyms update + 머신 저장)
export async function updateAdminGymFull(
  id: string, name: string, location: string, phone: string, hours: string, machineIds: string[],
): Promise<{ error: string | null }> {
  try {
    const { geocodeAddress } = await import('../geocoding');
    const coords = await geocodeAddress(location.trim());

    const { error: updateError } = await supabase
      .from('gyms')
      .update({
        name:      name.trim(),
        location:  location.trim(),
        phone:     phone.trim(),
        hours:     hours.trim(),
        latitude:  coords?.latitude  ?? null,
        longitude: coords?.longitude ?? null,
      })
      .eq('id', id);
    if (updateError) return { error: updateError.message };

    // 보유 머신 갱신 (전체 교체)
    const { error: machineError } = await saveGymMachineIds(id, machineIds);
    if (machineError) return { error: machineError };

    invalidateGymsCache();
    return { error: null };
  } catch (e: any) {
    return { error: e.message ?? '알 수 없는 오류' };
  }
}
