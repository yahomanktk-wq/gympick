## STEP 1: KCC-Ganpan 폰트 useFonts에 추가

`src/screens/flow/PostWorkoutLogModal.tsx` 에서
`useFonts` 훅을 찾아서 `KCCGanpan` 항목을 추가한다.

기존:
```tsx
const [fontsLoaded] = useFonts({
  'MatchaCih': require('../../../assets/fonts/Matcha_Cih.ttf'),
  'MatSaleh':  require('../../../assets/fonts/Mat_Saleh.ttf'),
});
```

변경 후:
```tsx
const [fontsLoaded] = useFonts({
  'MatchaCih':  require('../../../assets/fonts/Matcha_Cih.ttf'),
  'MatSaleh':   require('../../../assets/fonts/Mat_Saleh.ttf'),
  'KCCGanpan':  require('../../../assets/fonts/KCC-Ganpan.ttf'),
});
```

## STEP 2: CenteredOverlay 컴포넌트 수정

`src/screens/flow/PostWorkoutLogModal.tsx` 에서
`CenteredOverlay` 함수 전체를 찾아서 아래 코드로 교체한다.

```tsx
export function CenteredOverlay({ exercises, logs, nickname, timestamp, muscleLabel, tc, photoW, photoH }: OverlayProps) {
  const COL_COUNT = 2;
  const COL_W = Math.floor((photoW - OVERLAY_PAD * 2 - 12) / COL_COUNT);
  return (
    <>
      {/* 상단 좌: 닉네임 - KCC-Ganpan 폰트 */}
      <View style={{ position: 'absolute', top: OVERLAY_PAD, left: OVERLAY_PAD }}>
        <Text style={{ fontSize: 13, color: tc, letterSpacing: 0.3, fontFamily: 'KCCGanpan' }}>
          {nickname}
        </Text>
      </View>
      {/* 상단 우: 날짜 */}
      <View style={{ position: 'absolute', top: OVERLAY_PAD, right: OVERLAY_PAD, alignItems: 'flex-end' }}>
        <Text style={{ fontSize: 10, color: tc, opacity: 0.85, fontFamily: 'MatSaleh' }}>
          {timestamp}
        </Text>
      </View>
      {/* 중앙 타이틀 - Matcha_Cih 폰트, 밑줄 없음 */}
      <View style={{ position: 'absolute', top: OVERLAY_PAD + 44, left: OVERLAY_PAD, right: OVERLAY_PAD, alignItems: 'center' }}>
        <Text style={{
          fontSize: 32,
          color: tc,
          letterSpacing: -0.5,
          textAlign: 'center',
          fontFamily: 'MatchaCih',
        }}>
          {muscleLabel}
        </Text>
      </View>
      {/* 2열 운동 목록 - Mat_Saleh 폰트 */}
      <View style={{
        position: 'absolute',
        top: OVERLAY_PAD + 44 + 52,
        left: OVERLAY_PAD,
        right: OVERLAY_PAD,
        flexDirection: 'row',
        flexWrap: 'wrap',
        gap: 8,
        alignContent: 'flex-start',
      }}>
        {exercises.slice(0, 8).map(ex => {
          const sets = logs[ex.id] ?? [];
          const setsLine = sets.map(st => `${st.weight} × ${st.reps}`).join(' / ');
          return (
            <View key={ex.id} style={{ width: COL_W, marginBottom: 10 }}>
              <Text style={{
                fontSize: 9,
                color: tc,
                letterSpacing: 0.3,
                marginBottom: 2,
                fontFamily: 'MatSaleh',
              }} numberOfLines={1}>
                {`[${ex.brand.toUpperCase()}] ${ex.name.toUpperCase()}`}
              </Text>
              {setsLine ? (
                <Text style={{
                  fontSize: 8,
                  color: tc,
                  opacity: 0.85,
                  fontFamily: 'MatSaleh',
                }} numberOfLines={1}>
                  {setsLine}
                </Text>
              ) : null}
            </View>
          );
        })}
        {exercises.length > 8 && (
          <Text style={{ fontSize: 8, color: tc, opacity: 0.6, fontFamily: 'MatSaleh' }}>
            {`+${exercises.length - 8} more...`}
          </Text>
        )}
      </View>
      {/* 하단 중앙 로고 - 기존 0.85에서 30% 추가 축소 → 0.595 */}
      <View style={{ position: 'absolute', bottom: OVERLAY_PAD, left: 0, right: 0, alignItems: 'center' }}>
        <GymPickLogo color={tc} width={Math.round(LOGO_STRIP_W * 0.595)} height={Math.round(LOGO_STRIP_H * 0.595)} />
      </View>
    </>
  );
}
```
