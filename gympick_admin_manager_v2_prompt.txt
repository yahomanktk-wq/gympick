import { supabase } from '../supabase'

export async function submitManagerRequest(request: {
  userId: string; userName: string; userEmail: string
  gymName: string; gymLocation: string; contactPhone: string
  businessNumber: string; businessLicenseUrl?: string
  gymPhotoUrl?: string; operatingHours?: string
}): Promise<void> {
  const { error } = await supabase.from('manager_requests').insert({
    user_id: request.userId, user_name: request.userName,
    user_email: request.userEmail, gym_name: request.gymName,
    gym_location: request.gymLocation, contact_phone: request.contactPhone,
    business_number: request.businessNumber,
    business_license_url: request.businessLicenseUrl,
    gym_photo_url: request.gymPhotoUrl,
    operating_hours: request.operatingHours, status: 'pending',
  })
  if (error) throw error
}

export async function getManagerRequests(): Promise<any[]> {
  const { data, error } = await supabase
    .from('manager_requests').select('*')
    .order('created_at', { ascending: false })
  if (error) throw error
  return data ?? []
}

export async function approveManagerRequest(
  requestId: string, userId: string, gymId: string
): Promise<void> {
  const { error: e1 } = await supabase
    .from('manager_requests').update({ status: 'approved' }).eq('id', requestId)
  if (e1) throw e1
  const { error: e2 } = await supabase
    .from('user_profiles')
    .update({ role: 'manager', managed_gym_ids: [gymId] }).eq('id', userId)
  if (e2) throw e2
}

export async function rejectManagerRequest(requestId: string, reason: string): Promise<void> {
  const { error } = await supabase
    .from('manager_requests')
    .update({ status: 'rejected', reject_reason: reason }).eq('id', requestId)
  if (error) throw error
}

import { supabase } from '../supabase'
import { getAdminUserIds } from './userProfiles'

export async function getNotifications(userId: string): Promise<any[]> {
  const { data } = await supabase
    .from('notifications').select('*').eq('user_id', userId)
    .order('created_at', { ascending: false })
  return data ?? []
}

export async function markNotificationRead(id: string): Promise<void> {
  await supabase.from('notifications').update({ read: true }).eq('id', id)
}

export async function markAllNotificationsRead(userId: string): Promise<void> {
  await supabase.from('notifications').update({ read: true }).eq('user_id', userId)
}

export async function sendNotification(
  userId: string, title: string, message: string, type = 'info'
): Promise<void> {
  await supabase.from('notifications')
    .insert({ user_id: userId, title, message, type, read: false })
}

export async function sendNotificationToMultiple(
  userIds: string[], title: string, message: string, type = 'info'
): Promise<void> {
  if (userIds.length === 0) return
  const rows = userIds.map(userId => ({ user_id: userId, title, message, type, read: false }))
  await supabase.from('notifications').insert(rows)
}

export async function getUnreadCount(userId: string): Promise<number> {
  const { count } = await supabase
    .from('notifications').select('*', { count: 'exact', head: true })
    .eq('user_id', userId).eq('read', false)
  return count ?? 0
}

export async function notifyMachineChanges(
  gymId: string,
  gymName: string,
  addedMachineIds: string[],
  removedMachineIds: string[]
): Promise<void> {
  if (addedMachineIds.length === 0 && removedMachineIds.length === 0) return

  const affectedIds = [...addedMachineIds, ...removedMachineIds]
  const { data: favorites } = await supabase
    .from('favorite_machines').select('user_id').in('machine_id', affectedIds)

  const favoriteUserIds = [...new Set((favorites ?? []).map((f: any) => f.user_id))]
  const adminIds = await getAdminUserIds()
  const allRecipients = [...new Set([...favoriteUserIds, ...adminIds])]
  if (allRecipients.length === 0) return

  const parts: string[] = []
  if (addedMachineIds.length > 0) parts.push(`머신 ${addedMachineIds.length}개 추가`)
  if (removedMachineIds.length > 0) parts.push(`머신 ${removedMachineIds.length}개 제거`)

  await sendNotificationToMultiple(
    allRecipients,
    `${gymName} 머신 변경 안내`,
    `${gymName}에서 ${parts.join(', ')}되었습니다.`,
    'machine'
  )
}

import { supabase } from '../supabase'

export async function getUserProfile(userId: string): Promise<any | null> {
  const { data } = await supabase
    .from('user_profiles').select('*').eq('id', userId).single()
  return data ?? null
}

export async function upsertUserProfile(profile: {
  id: string; email: string; name: string; nickname?: string; role?: string
}): Promise<void> {
  await supabase.from('user_profiles').upsert({
    id: profile.id, email: profile.email, name: profile.name,
    nickname: profile.nickname, role: profile.role ?? 'user',
  })
}

export async function getAllUsers(): Promise<any[]> {
  const { data } = await supabase
    .from('user_profiles').select('*').order('created_at', { ascending: false })
  return data ?? []
}

export async function getAdminUserIds(): Promise<string[]> {
  const { data } = await supabase
    .from('user_profiles').select('id').eq('role', 'admin')
  return (data ?? []).map((u: any) => u.id)
}

export async function getGymManagerIds(gymId: string): Promise<string[]> {
  const { data } = await supabase
    .from('user_profiles').select('id').contains('managed_gym_ids', [gymId])
  return (data ?? []).map((u: any) => u.id)
}

export async function updateUserRole(
  userId: string, role: string, managedGymIds?: string[]
): Promise<void> {
  await supabase.from('user_profiles')
    .update({ role, managed_gym_ids: managedGymIds ?? [] }).eq('id', userId)
}

import { supabase } from '../supabase'

export type ReportType =
  | 'machine_missing'
  | 'machine_broken'
  | 'info_wrong'
  | 'gym_closed'
  | 'other'

export const REPORT_TYPE_LABELS: Record<ReportType, string> = {
  machine_missing: '머신이 없어요',
  machine_broken:  '머신이 고장났어요',
  info_wrong:      '정보가 달라요',
  gym_closed:      '폐업했어요',
  other:           '기타',
}

export async function submitGymReport(report: {
  gymId: string; userId: string; reportType: ReportType
  description: string; machineId?: string
}): Promise<void> {
  const { error } = await supabase.from('gym_reports').insert({
    gym_id: report.gymId, user_id: report.userId,
    report_type: report.reportType, description: report.description,
    machine_id: report.machineId, status: 'pending',
  })
  if (error) throw error
}

export async function getGymReports(status?: string): Promise<any[]> {
  let query = supabase.from('gym_reports').select('*')
    .order('created_at', { ascending: false })
  if (status) query = query.eq('status', status)
  const { data } = await query
  return data ?? []
}

export async function resolveGymReport(reportId: string): Promise<void> {
  await supabase.from('gym_reports').update({ status: 'resolved' }).eq('id', reportId)
}
