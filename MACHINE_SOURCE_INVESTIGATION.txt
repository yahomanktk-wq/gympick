function rowToMachine(row: MachineRow): Machine {
  return {
    id:           row.id,
    name:         row.name,
    brand:        row.brand,
    targetMuscle: row.target_muscle,   // snake → camel
    category:     row.category,
    exerciseKind: row.exercise_kind as ExerciseKind,  // snake → camel
    imageUrl:     row.image_url ?? undefined,          // snake → camel, null → undefined
  };
}
