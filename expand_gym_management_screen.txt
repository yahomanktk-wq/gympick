STEP 2 — 매니저 연결 (link a manager to a gym by email/nickname search)
============================================================
1. Add a new lib function in src/lib/db/adminGyms.ts (or a suitable existing
   db module):

    export interface ProfileSearchResult {
      id: string;
      email: string | null;
      nickname: string | null;
    }

    export async function searchProfilesByEmailOrNickname(query: string): Promise<ProfileSearchResult[]> {
      const q = query.trim();
      if (!q) return [];
      const { data, error } = await supabase
        .from('profiles')
        .select('id, email, nickname')
        .or(`email.ilike.%${q}%,nickname.ilike.%${q}%`)
        .limit(20);
      if (error) {
        if (__DEV__) console.warn('[db/adminGyms] searchProfilesByEmailOrNickname error:', error.message);
        return [];
      }
      return data ?? [];
    }

    export async function setGymManager(gymId: string, managerId: string | null): Promise<void> {
      const { error } = await supabase
        .from('gyms')
        .update({ manager_id: managerId })
        .eq('id', gymId);
      if (error) throw new Error(error.message);
    }
